cmake_minimum_required(VERSION 3.21)
project(ClassicMac VERSION 4.6.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Network Concurrent)
find_package(ZLIB REQUIRED)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
add_executable(classicmac WIN32 src/main.cpp src/platinum.cpp src/widgets.cpp src/machine.cpp src/catalog.cpp src/engine.cpp src/vnc.cpp src/app.cpp src/qualification.cpp resources/assets.qrc)
target_link_libraries(classicmac PRIVATE Qt6::Widgets Qt6::Network Qt6::Concurrent ZLIB::ZLIB OpenSSL::Crypto)
target_compile_definitions(classicmac PRIVATE CLASSICMAC_VERSION="${PROJECT_VERSION}")
if(WIN32)
  target_sources(classicmac PRIVATE resources/classicmac.rc)
  target_link_libraries(classicmac PRIVATE ws2_32)
endif()
if(MSVC)
  target_compile_options(classicmac PRIVATE /W4 /utf-8)
else()
  target_compile_options(classicmac PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME native-tests COMMAND classicmac --self-test ${CMAKE_CURRENT_SOURCE_DIR}/../app/Tests/ClassicMacTests/Fixtures/Platinum)
set_tests_properties(native-tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
install(TARGETS classicmac RUNTIME DESTINATION bin)
